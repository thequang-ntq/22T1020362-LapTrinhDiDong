    var getUrl = "https://dummyjson.com/auth/me";

    try {
      var response = await dio.request(getUrl);
      
      if (response.statusCode == 200) {
        List data = response.data['articles'];
        ls = data.map((json) => News.fromJson(json)).toList();
      } else {
        String error = response.statusMessage ?? "Lỗi không xác định";
        print("Lỗi: $error");
      }
    } catch (e) {
      print("Lỗi kết nối: $e");
    }
    
    return ls;
  }
}

class News {
  final String title;
  final String description;
  final String url;
  final String urlToImage;
  final String publishedAt;
  final String author;
  final String content;
  final Source source;

  News({
    required this.title,
    required this.description,
    required this.url,
    required this.urlToImage,
    required this.publishedAt,
    required this.author,
    required this.content,
    required this.source,
  });

  factory News.fromJson(Map<String, dynamic> json) {
    return News(
      title: json['title'] ?? 'No Title',
      description: json['description'] ?? 'No Description',
      url: json['url'] ?? '',
      urlToImage: json['urlToImage'] ?? '../../assets/images/placeholder.jpg',
      publishedAt: json['publishedAt'] ?? '',
      author: json['author'] ?? 'Unknown',
      content: json['content'] ?? 'No Content',
      source: Source.fromJson(json['source'] ?? {}),
    );
  }
}

class Source {
  final String id;
  final String name;

  Source({
    required this.id,
    required this.name,
  });

  factory Source.fromJson(Map<String, dynamic> json) {
    return Source(
      id: json['id'] ?? '',
      name: json['name'] ?? 'Unknown Source',
    );
  }
}

* Yêu cầu:
- Hãy sửa lại các file trên và bổ sung các file mới theo yêu cầu tạo trang web sau:
+ Ban đầu khi bật ứng dụng sẽ mở ra trang đăng nhập, Trong trang đăng nhập sẽ có AppBar với tiêu đề là "22T1020362-Đăng nhập", sau đó phần Body sẽ là form đăng nhập với 2 label "Tên đăng nhập" và "Mật khẩu" cùng 2 TextFormField tương ứng, ở dưới cùng là nút Đăng nhập. Lưu ý 2 TextFormField trên sẽ có Controller, bên Tên đăng nhập cần kiểm tra khác rỗng, bên Mật khẩu cần kiểm tra khác rỗng và có ít nhất 6 ký tự (Giống phần kiểm tra trong api_login_form). Xử lý dữ liệu trên URL post ở trên, lấy accessToken, lưu để đăng nhập vào lấy chi tiết người dùng ở URL get, xóa dữ liệu đã nhập trong trang đăng nhập.
+ Khi bấm vào nút Đăng nhập, sẽ hiển thị trang mới là trang Chi tiết người dùng, lấy dữ liệu trên URL get ở trên. Hiển thị tất cả dữ liệu cho đẹp. Ở dưới cùng là nút Đăng xuất, khi bấm vào sẽ xóa accessToken và quay lại trang Đăng nhập.
